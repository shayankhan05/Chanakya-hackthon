import type {Workspace} from './domain';
import {legalRules} from './legal-catalogue';
import {simulate} from './engine';
export function retrieveRules(question:string){const q=question.toLowerCase();const keys:Record<string,string[]>={POSH:['posh','harass','committee'],EPF:['epf','provident','hire','hiring','employee'],ESI:['esi','insurance','wage'],GST:['gst','tax','turnover'],UDYAM:['msme','udyam','investment','classif'],IEC:['export','iec'],PT:['professional tax'],SHOP:['shop','branch','state','workplace'],LICENSE:['license','renew','expiry']};const selected=legalRules.filter(r=>keys[r.id]?.some(k=>q.includes(k)));return selected.length?selected:legalRules.filter(r=>['EPF','POSH','GST'].includes(r.id));}
export function knowledgeAnswer(question:string,w:Workspace){const q=question.toLowerCase();const relevant=retrieveRules(q);let content='';let links=[{label:'Review business findings',view:'laws'}];const hiring=/hire|hiring|employee|people/.test(q);if(hiring&&/gst|tax/.test(q)){content='Hiring alone does not determine GST registration or increase GST payable. GST depends on turnover, supply type, state, taxable value, rates, exemptions and input tax credit.\n\nYour GST review: '+w.findings.find(f=>f.ruleId==='GST')?.reason+'\n\nThe app does not calculate GST payable. Payroll changes are evaluated separately for EPF, ESI and POSH.';}
 else if(hiring&&/if|more|plan|additional|reduce/.test(q)){const match=q.match(/(?:hire|hiring|add|reduce|remove)\s*(\d+)|(?:\b)(\d+)\s*(?:more|additional|fewer)/);if(match){const n=Number(match[1]||match[2]);const count=Math.max(0,w.profile.employees+(/reduce|remove|fewer/.test(q)?-n:n));const planned={...structuredClone(w.profile),employees:count,eligibleEsiEmployees:w.profile.eligibleEsiEmployees===null?null:Math.min(count,w.profile.eligibleEsiEmployees)};if(planned.branchLocations.length===1)planned.branchLocations[0].employees=count;const result=simulate(w,planned);content=`Planning only: employee count ${w.profile.employees} → ${count}. Your saved profile is unchanged.\n\n`+result.impacts.filter(i=>['EPF','ESI','POSH'].includes(i.ruleId)).map(i=>`${i.name}: ${i.after}. ${i.reason}`).join('\n\n')+'\n\nGST was evaluated independently; hiring does not change turnover or tax payable.';}else content='Enter the number of planned hires or reductions in Plan your business growth. The simulator compares all relevant domains without changing your saved business.';links=[{label:'Open growth simulator',view:'simulator'}];}
 else if(/document|evidence|prepare|steps/.test(q)&&/follow-up topic/i.test(q)){content=relevant.map(r=>`${r.plainName}\n\nDocuments and evidence to prepare: ${r.documents.join(', ')}.\n\n${r.actions.map((a,i)=>`${i+1}. ${a}`).join('\n')}\n\nFor your business: ${w.findings.find(f=>f.ruleId===r.id)?.reason||'Confirm the business facts first.'}`).join('\n\n');}
 else if(/task|pending|prioriti|risk|next|do first/.test(q)&&!/what is|what does/.test(q)){const task=w.tasks.find(t=>t.status!=='Completed'&&relevant.some(r=>r.id===t.ruleId))||w.tasks.find(t=>t.status!=='Completed');content=task?`Your next task: ${task.title}\n\n${task.description}\n\nPrepare: ${task.documents.join(', ')}.\nRecommended target: ${task.targetDate||'not set'} (planning date).\nLegal due date: ${task.legalDueDate||'not verified or recorded'}.\n\nOpen the task to assign a responsible person, record evidence, and update progress.`:'There are no unfinished tasks in this workspace. Review missing information and new business changes before treating the business as fully ready.';links=[{label:'Open action tasks',view:'tasks'}];}
 else if(/epf|posh|esi|gst|udyam|msme|iec|export|professional tax|shop|licen/.test(q)){content=relevant.map(r=>{const f=w.findings.find(f=>f.ruleId===r.id);return `${r.plainName}\n${r.knowledge}\n\nFor your business: ${f?.outcome||'Not yet evaluated'}. ${f?.reason||''}\nFacts evaluated: ${f?.facts.join('; ')||'See business profile.'}${f?.missing.length?'\nInformation still needed: '+f.missing.join(', ')+'.':''}\n\nNext steps:\n${r.actions.map((a,i)=>`${i+1}. ${a}`).join('\n')}`;}).slice(0,2).join('\n\n');}
 else content='This limited knowledge-base search covers EPF, ESI, POSH, GST, professional tax, workplace registration, MSME classification, exports and license reminders. It is not a live AI response.\n\nTry “What is POSH?”, “Why is EPF showing for my company?”, “If I hire 5 more people, what happens?” or open a task for a contextual explanation.';
 return {message:content,mode:'knowledge-base' as const,sources:relevant.filter(r=>r.sourceUrl).map(r=>({name:r.plainName+' · '+r.reference,url:r.sourceUrl})),links};}
