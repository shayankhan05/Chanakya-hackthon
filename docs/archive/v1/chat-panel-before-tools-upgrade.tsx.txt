'use client';
import {useEffect,useRef,useState} from 'react';
import {Send,Sparkles} from 'lucide-react';
import type {Workspace,ChatMessage} from '@/lib/domain';
import Modal from './modal';
import {Button} from './ui/button';
export default function ChatPanel({workspace,onWorkspace,onClose,navigate,prompt=''}:{workspace:Workspace;onWorkspace:(w:Workspace)=>void;onClose:()=>void;navigate:(v:string)=>void;prompt?:string}){const [messages,setMessages]=useState(workspace.messages),[question,setQuestion]=useState(prompt),[busy,setBusy]=useState(false),[error,setError]=useState(''),[warning,setWarning]=useState(''),[mode,setMode]=useState('Source search available; live AI requires server configuration.');const sending=useRef(false);const scroll=useRef<HTMLDivElement>(null);useEffect(()=>{fetch('/api/chat').then(r=>r.json()).then(d=>setMode(workspace.demo?'Limited knowledge-base search · not a live AI response':d.liveConfigured?'Live AI configured · connection checked on each question':'Limited knowledge-base search · OPENAI_API_KEY is not configured')).catch(()=>setMode('Assistant configuration could not be loaded. Try sending a question.'));},[workspace.demo]);useEffect(()=>{scroll.current?.scrollIntoView({block:'nearest'})},[messages,busy]);async function send(text=question){if(sending.current||!text.trim())return;sending.current=true;setBusy(true);setError('');setWarning('');const id=crypto.randomUUID(),at=new Date().toISOString();const user:ChatMessage={id,role:'user',content:text,at};setMessages(m=>[...m,user]);setQuestion('');try{const r=await fetch('/api/chat',{method:'POST',signal:AbortSignal.timeout(40000),headers:{'Content-Type':'application/json'},body:JSON.stringify({question:text,messageId:id,businessId:workspace.id,demoProfile:workspace.demo?workspace.profile:undefined,demoTasks:workspace.demo?workspace.tasks.map(t=>({id:t.id,title:t.title,ruleId:t.ruleId,description:t.description,status:t.status,documents:t.documents,targetDate:t.targetDate,legalDueDate:t.legalDueDate})):undefined,history:messages.slice(-20).map(m=>({role:m.role,content:m.content}))})});if(!r.headers.get('content-type')?.includes('application/json'))throw Error('The chat endpoint returned an unexpected response (HTTP '+r.status+'). Check the running server URL.');const d=await r.json();if(!r.ok)throw Error((d.error||'Chat request failed')+' (HTTP '+r.status+').');const answer:ChatMessage={id:id+'-answer',role:'assistant',content:d.message,at:new Date().toISOString(),mode:d.mode,sources:d.sources,links:d.links};setMessages(m=>[...m,answer]);setMode(d.mode==='live'?'Live AI · grounded in the checked source catalogue':'Limited knowledge-base search · not a live AI response');setWarning(d.providerError||d.configuration||'');if(d.workspace)onWorkspace(d.workspace);else onWorkspace({...workspace,messages:[...messages,user,answer].slice(-200)});}catch(e){setError(e instanceof Error&&e.name==='TimeoutError'?'The chat request timed out. Check the backend connection and retry.':e instanceof TypeError?'Could not reach the chat backend. Check that the server is running at this page’s address.':e instanceof Error?e.message:'Network error. Please try again.');setQuestion(text);}finally{sending.current=false;setBusy(false)}}return <Modal title="Your compliance copilot" onClose={onClose} wide><p className="text-[11px] text-indigo-600 mb-3 flex gap-2 items-center"><Sparkles size={14}/>{mode}</p><div className="h-[48vh] min-h-48 overflow-y-auto bg-slate-50 rounded-xl p-4 space-y-4">{messages.length===0&&<div className="text-sm text-slate-500 leading-7">Ask about your business findings, a pending task, or a possible growth step. Try “What is POSH?” or “If I hire 5 more people, what happens?”</div>}{messages.map(m=><div key={m.id} className={`p-4 rounded-xl text-xs leading-6 ${m.role==='user'?'bg-indigo-600 text-white ml-8':'bg-white border border-slate-100 mr-8'}`}><p className="whitespace-pre-wrap">{m.content}</p>{m.mode&&<p className="text-[9px] text-slate-400 mt-3">{m.mode==='live'?'AI-generated explanation':'Limited source-based explanation'}</p>}{m.sources&&<div className="mt-3 space-y-1">{m.sources.map(s=><a target="_blank" rel="noreferrer" key={s.url+s.name} href={s.url} className="block text-[10px] text-indigo-600 underline">{s.name} ↗</a>)}</div>}{m.links&&<div className="flex flex-wrap gap-2 mt-3">{m.links.map(l=><button key={l.view} className="tag bg-indigo-50 text-indigo-600" onClick={()=>{navigate(l.view);onClose()}}>{l.label} →</button>)}</div>}</div>)}{busy&&<p role="status" className="text-xs text-indigo-500 animate-pulse">Checking your question and source context…</p>}<div ref={scroll}/></div>{warning&&<p role="status" className="text-[11px] text-amber-700 bg-amber-50 rounded p-2 mt-3">{warning}</p>}{error&&<p role="alert" className="text-xs text-rose-600 mt-3">{error}</p>}<div className="flex flex-wrap gap-2 my-4">{['What is POSH?','Why is EPF showing for my company?','If I hire 5 more people, what happens?'].map(q=><button disabled={busy} key={q} onClick={()=>send(q)} className="tag bg-indigo-50 text-indigo-600 disabled:opacity-40">{q}</button>)}</div><form onSubmit={e=>{e.preventDefault();send()}} className="flex gap-2"><input aria-label="Your compliance question" className="field" maxLength={2000} placeholder="Ask a question about your business…" value={question} onChange={e=>setQuestion(e.target.value)}/><Button disabled={busy||!question.trim()} aria-label="Send question"><Send size={16}/></Button></form><p className="text-[10px] text-slate-400 mt-3">Guidance supports a decision; verify uncertain legal facts with a qualified professional.</p></Modal>}
