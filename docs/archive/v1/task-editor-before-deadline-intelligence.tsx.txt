'use client';
import {useState} from 'react';
import type {Workspace,WorkTask} from '@/lib/domain';
import type {Mutation} from '@/lib/mutations';
import Modal from './modal';
import {Button} from './ui/button';
export default function TaskEditor({task,workspace,mutate,onClose,onChat,onWorkspace}:{task:WorkTask;workspace:Workspace;mutate:(m:Mutation)=>Promise<void>;onClose:()=>void;onChat:()=>void;onWorkspace:(w:Workspace)=>void}){const [draft,setDraft]=useState({...task,legalDueDate:task.legalDueDate||'',note:''}),[busy,setBusy]=useState(false),[error,setError]=useState('');const finding=workspace.findings.find(f=>f.ruleId===task.ruleId);return <Modal title={task.title} onClose={onClose}><span className="tag bg-indigo-50 text-indigo-600">{finding?.name||task.ruleId}</span><p className="text-xs leading-6 text-slate-600 mt-4">{task.description}</p><h3 className="font-semibold text-sm mt-5">Your next steps</h3><ol className="list-decimal ml-5 text-xs leading-6 text-slate-600 mt-2">{finding?.actions.map(a=><li key={a}>{a}</li>)}</ol><p className="text-xs text-slate-500 mt-4">Evidence to prepare: {task.documents.join(', ')}</p>{finding?.missing.length? <p className="text-xs bg-amber-50 text-amber-800 p-3 rounded-lg mt-4">Still to confirm: {finding.missing.join(', ')}</p>:null}<form onSubmit={async e=>{e.preventDefault();setBusy(true);setError('');try{await mutate({type:'task',id:task.id,status:draft.status,assignee:draft.assignee,targetDate:draft.targetDate,legalDueDate:draft.legalDueDate,sourceReference:draft.sourceReference,note:draft.note});onClose();}catch(e){setError(e instanceof Error?e.message:'Could not save.')}finally{setBusy(false)}}}><div className="grid sm:grid-cols-2 gap-4 mt-5"><label className="text-xs">Status<select className="field mt-2" value={draft.status} onChange={e=>setDraft({...draft,status:e.target.value as WorkTask['status']})}>{['Pending','In progress','Completed','Overdue','Needs review'].map(s=><option key={s}>{s}</option>)}</select></label><label className="text-xs">Responsible person<input className="field mt-2" maxLength={100} value={draft.assignee} onChange={e=>setDraft({...draft,assignee:e.target.value})}/></label><label className="text-xs">Preparation target<input type="date" className="field mt-2" value={draft.targetDate} onChange={e=>setDraft({...draft,targetDate:e.target.value})}/><span className="text-[10px] text-slate-400">Planning date, not a legal deadline.</span></label><label className="text-xs">Legal due date if verified<input type="date" className="field mt-2" value={draft.legalDueDate} onChange={e=>setDraft({...draft,legalDueDate:e.target.value})}/><span className="text-[10px] text-slate-400">User-recorded; requires a supporting reference.</span></label><label className="sm:col-span-2 text-xs">Source reference for deadline<input className="field mt-2" maxLength={500} value={draft.sourceReference} onChange={e=>setDraft({...draft,sourceReference:e.target.value})}/></label><label className="sm:col-span-2 text-xs">Completion or evidence note<textarea className="field mt-2" maxLength={500} value={draft.note} onChange={e=>setDraft({...draft,note:e.target.value})}/></label></div><div className="mt-5 text-xs text-slate-500"><h4 className="font-semibold mb-2">Saved evidence</h4>{task.evidence.length===0&&<p>No evidence recorded.</p>}{task.evidence.map((ev,i)=>{const doc=workspace.documents.find(d=>d.id===ev);return doc?<a className="block text-indigo-600 py-1" href={`/api/workspaces/${workspace.id}/evidence/${doc.id}`} key={ev}>{doc.name} ↓</a>:<p className="py-1" key={i}>{ev}</p>})}{workspace.demo?<p className="mt-3 text-[11px] text-slate-400">Demo supports evidence notes. Sign in to upload PDF or text evidence.</p>:<label className="block mt-3 text-indigo-600 cursor-pointer">Upload PDF or text evidence up to 2 MB<input aria-label="Upload evidence" className="block mt-2 text-xs" type="file" accept="application/pdf,text/plain" disabled={busy} onChange={async e=>{const file=e.target.files?.[0];if(!file)return;setBusy(true);setError('');try{const body=new FormData();body.set('file',file);body.set('taskId',task.id);const r=await fetch(`/api/workspaces/${workspace.id}/evidence`,{method:'POST',body});const d=await r.json();if(!r.ok)throw Error(d.error);onWorkspace(d.workspace);onClose();}catch(e){setError(e instanceof Error?e.message:'Upload failed.')}finally{setBusy(false)}}}/></label>}</div><p className="text-[11px] leading-5 text-slate-500 mt-5">{finding?.consequence}</p>{task.sourceUrl&&<a className="text-xs text-indigo-600 mt-2 block" href={task.sourceUrl} target="_blank" rel="noreferrer">Supporting source ↗</a>}{error&&<p role="alert" className="text-xs text-rose-600 mt-4">{error}</p>}<div className="flex justify-end gap-2 mt-6"><Button type="button" variant="outline" onClick={onChat}>Explain with Copilot</Button><Button disabled={busy} type="submit">{busy?'Saving…':'Save task progress'}</Button></div></form></Modal>}
