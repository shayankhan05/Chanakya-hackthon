import {NextResponse} from 'next/server';
import {getServerSession} from 'next-auth';
import {authOptions} from '@/lib/auth';
import {db,demoMode} from '@/lib/db';
import {z} from 'zod';
export async function PATCH(req:Request){try{const data=z.object({id:z.string().min(1),status:z.enum(['Pending','In progress','Completed'])}).safeParse(await req.json());if(!data.success)return NextResponse.json({error:'Invalid task'},{status:400});if(demoMode)return NextResponse.json({success:true,demo:true});const session=await getServerSession(authOptions);if(!session?.user?.email)return NextResponse.json({error:'Unauthorized'},{status:401});const result=await db.complianceTask.updateMany({where:{id:data.data.id,profile:{user:{email:session.user.email}}},data:{status:data.data.status}});return NextResponse.json({success:result.count>0},{status:result.count?200:404})}catch{return NextResponse.json({error:'Unable to update task'},{status:500})}}
