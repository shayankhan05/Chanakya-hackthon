import {NextResponse} from 'next/server';
import {getServerSession} from 'next-auth';
import {authOptions} from '@/lib/auth';
import {db,demoMode} from '@/lib/db';
import {profileSchema} from '@/lib/validation';
import {evaluate,type Rule} from '@/lib/rules';
import {initialTasks} from '@/lib/data';
export async function GET(){if(demoMode)return NextResponse.json({demo:true});const session=await getServerSession(authOptions);if(!session?.user?.email)return NextResponse.json({error:'Unauthorized'},{status:401});const user=await db.user.findUnique({where:{email:session.user.email},include:{profile:{include:{tasks:true,alerts:true,licenses:true,registrations:true,applicableLaws:true}}}});return NextResponse.json(user?.profile??null)}
export async function POST(req:Request){try{const parsed=profileSchema.safeParse(await req.json());if(!parsed.success)return NextResponse.json({error:'Invalid business profile'},{status:400});const profile=parsed.data;if(demoMode)return NextResponse.json({profile,applicable:evaluate(profile),demo:true});const session=await getServerSession(authOptions);if(!session?.user?.email)return NextResponse.json({error:'Unauthorized'},{status:401});const user=await db.user.upsert({where:{email:session.user.email},create:{email:session.user.email,name:session.user.name},update:{}});const stored=await db.complianceRule.findMany({where:{enabled:true}});const applicable=evaluate(profile,stored.map(r=>({code:r.code,label:r.label,field:r.field,operator:r.operator,value:r.value,reason:r.reason})) as Rule[]);const {registrations,...fields}=profile;const result=await db.$transaction(async tx=>{const p=await tx.sMEProfile.upsert({where:{userId:user.id},create:{...fields,userId:user.id},update:fields});await tx.applicableLaw.deleteMany({where:{profileId:p.id}});await tx.applicableLaw.createMany({data:applicable.map(l=>({profileId:p.id,code:l.code,name:l.label,reason:l.reason}))});await tx.registration.deleteMany({where:{profileId:p.id}});await tx.registration.createMany({data:registrations.map(name=>({profileId:p.id,name,status:'Registered'}))});for(const law of applicable){const id=p.id+'-'+law.code;const template=initialTasks.find(t=>t.law===law.label);await tx.complianceTask.upsert({where:{id},create:{id,profileId:p.id,title:template?.title||'Review '+law.label,law:law.label,due:new Date(Date.now()+7*86400000),priority:template?.priority||'Medium',status:'Pending',description:law.reason,documents:template?.documents||['Business identity documents'],penalty:template?.penalty||'Requires case-specific review.',duration:template?.duration||'1–3 working days'},update:{description:law.reason}})}return p});return NextResponse.json({profile:result,applicable})}catch{return NextResponse.json({error:'Unable to save profile'},{status:500})}}
